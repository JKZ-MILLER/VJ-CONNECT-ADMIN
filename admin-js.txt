function persistDraft(){
 try{localStorage.setItem("vjconnect_editor_draft",JSON.stringify({state,currentId}));$("statDraft").textContent="Salvo";}catch(e){}
}

function restoreDraft(){
 try{
  const raw=localStorage.getItem("vjconnect_editor_draft");
  if(!raw)return false;
  const d=JSON.parse(raw);
  if(!d?.state)return false;
  state={...clone(DEFAULT),...d.state,links:Array.isArray(d.state.links)?d.state.links:clone(DEFAULT.links)};
  currentId=d.currentId||null;
  return true;
 }catch(e){return false}
}

function clearDraft(){
 localStorage.removeItem("vjconnect_editor_draft");
 $("statDraft").textContent="—";
}

function addLink(icon){
 const labels={maps:"Localização",whatsapp:"WhatsApp",instagram:"Instagram",google:"Google",tiktok:"TikTok",facebook:"Facebook",youtube:"YouTube",link:"Novo link"};

function removeLink(i){
 if(!state.links[i])return;
 const label=state.links[i].label||"este link";
 if(!confirm('Apagar "'+label+'"?'))return;
 state.links.splice(i,1);
 renderLinks();render();persistDraft();
}

function updateLink(i,key,value){
 if(!state.links[i])return;
 state.links[i][key]=value;
 render();
 persistDraft();
}

function toggleLink(i){
 if(!state.links[i])return;
 state.links[i].enabled=state.links[i].enabled!==false?false:true;
 render();
 persistDraft();
}

function moveLink(i,dir){
 const j=i+dir;
 if(j<0||j>=state.links.length)return;
 [state.links[i],state.links[j]]=[state.links[j],state.links[i]];
 renderLinks();render();persistDraft();
}

function renderLinks(){

   </select>
   <input value="${esc(x.label)}" placeholder="Texto do botão" oninput="updateLink(${i},'label',this.value)">
   <input class="url" value="${esc(x.url)}" placeholder="https://..." oninput="updateLink(${i},'url',this.value)">
   <div class="row">
    <button class="btn icon" title="Mover para cima" onclick="moveLink(${i},-1)">↑</button>
    <button class="btn icon" title="Mover para baixo" onclick="moveLink(${i},1)">↓</button>

   </div>
  </div>`).join("")||'<div class="empty">Nenhum link. Use os botões abaixo para adicionar.</div>';
}

function visual(){

  ?`linear-gradient(${state.gradientDir},${state.bg},${state.bg2})`

  ?`linear-gradient(135deg,${state.button},${state.accent})`
  :state.buttonStyle==="outline"?"transparent":state.button;

 return {bg,button,border};
}

function normalizeUrl(v){
 const s=String(v||"").trim();
 if(!s)return "#";
 return /^https?:\/\//i.test(s)?s:"https://"+s;
}

async function uploadImage(kind,input){
 const file=input.files?.[0];
 if(!file)return;
 if(!file.type.startsWith("image/")){notify("Escolha uma imagem.");input.value="";return}
 if(file.size>5*1024*1024){notify("A imagem deve ter no máximo 5 MB.");input.value="";return}
 if(!db){notify("Supabase não configurado.");return}
 notify("Enviando imagem...");
 const ext=(file.name.split(".").pop()||"jpg").toLowerCase().replace(/[^a-z0-9]/g,"")||"jpg";
 const path=`${crypto.randomUUID()}.${ext}`;
 const up=await db.storage.from("client-assets").upload(path,file,{upsert:false,contentType:file.type});
 if(up.error){notify("Upload: "+up.error.message);input.value="";return}
 state[kind]=db.storage.from("client-assets").getPublicUrl(path).data.publicUrl;
 $(""+kind+"Url").value=state[kind];
 renderMedia();render();persistDraft();input.value="";
 notify("Imagem enviada.");
}

async function saveClient(){
 if(!db){notify("Supabase não configurado.");return}
 const p=payload();
 if(!p.name||!p.slug){notify("Informe nome e slug.");switchTab("content");return}
 notify("Salvando cliente...");
 let res=currentId
  ?await db.from("clients").update({...p,updated_at:new Date().toISOString()}).eq("id",currentId).select().single()
  :await db.from("clients").insert(p).select().single();

 if(res.error){notify("Erro ao salvar: "+res.error.message);return}
 currentId=res.data.id;

 const del=await db.from("client_links").delete().eq("client_id",currentId);
 if(del.error){notify("Cliente salvo, mas os links não foram atualizados: "+del.error.message);return}

 const rows=state.links.filter(x=>String(x.url||"").trim()).map((x,i)=>({
  client_id:currentId,position:i,label:String(x.label||"Link").trim()||"Link",
  url:normalizeUrl(x.url),icon_key:x.icon||"link",icon_url:null,enabled:x.enabled!==false
 }));
 if(rows.length){
  const lr=await db.from("client_links").insert(rows);
  if(lr.error){notify("Cliente salvo, mas houve erro nos links: "+lr.error.message);return}
 }
 persistDraft();
 $("formTitle").textContent="Editar Cliente";
 clearDraft();
 notify("✓ Cliente salvo. URL: "+publicURL(p.slug));
 await loadClients();
 showView("editor");
}

async function editClient(id){
 if(!db)return;
 notify("Carregando cliente...");
 const r=await db.from("clients").select("*").eq("id",id).single();
 if(r.error){notify("Erro: "+r.error.message);return}
 const c=r.data,t=c.theme||{};
 state={
  ...clone(DEFAULT),name:c.name||"",slug:c.slug||"",bio:c.bio||"",
  logo:c.logo_url||"",banner:c.banner_url||"",published:c.published!==false,
  bg:t.bg||DEFAULT.bg,bg2:t.bg2||DEFAULT.bg2,text:t.text||DEFAULT.text,button:t.button||DEFAULT.button,
  buttonText:t.buttonText||DEFAULT.buttonText,accent:t.accent||DEFAULT.accent,bgMode:t.bgMode||DEFAULT.bgMode,
  gradientDir:t.gradientDir||DEFAULT.gradientDir,buttonStyle:t.buttonStyle||DEFAULT.buttonStyle,
  buttonRadius:Number(t.buttonRadius??15),verifyText:t.verifyText||"",
  links:[]
 };
 const lr=await db.from("client_links").select("*").eq("client_id",id).order("position",{ascending:true});
 if(lr.error){notify("Erro nos links: "+lr.error.message);return}
 state.links=(lr.data||[]).map(x=>({label:x.label,url:x.url,icon:x.icon_key||"link",enabled:x.enabled!==false}));
 currentId=id;
 localStorage.setItem("vjconnect_editor_draft",JSON.stringify({state,currentId}));
 showView("editor");switchTab("content");notify("Cliente carregado.");
}

async function deleteClient(id){
 if(!db)return;
 const c=clients.find(x=>x.id===id);
 if(!confirm(`Excluir "${c?.name||"este cliente"}" e todos os links? Esta ação não pode ser desfeita.`))return;
 notify("Excluindo...");
 const r=await db.from("clients").delete().eq("id",id);
 if(r.error){notify("Erro ao excluir: "+r.error.message);return}
 if(currentId===id){currentId=null;newClient()}

 showView("clients");

}

async function boot(){

 const {data,error}=await db.auth.getSession();
 if(error){$("login").classList.remove("hidden");$("loginStatus").textContent=error.message;return}
 if(data.session) await enterApp(data.session); else $("login").classList.remove("hidden");
 db.auth.onAuthStateChange(async(_event,session)=>{
  if(session) await enterApp(session);
  else{$("app").classList.add("hidden");$("login").classList.remove("hidden")}
 });
}

 if(restored){
  applyStateToForm();
  $("statDraft").textContent="Salvo";
 }else newClient();
 await loadClients();
 showView("dashboard");
}
